# Docker - PostgreSQL - Consultorio

## Crear la imagen
docker build -t consultorio-pg .

## Crear el contenedor
docker run --name consultorio-db -e POSTGRES_PASSWORD=1234 -p 5432:5432 -d consultorio-pg

## Verificar que el contenedor esté corriendo
docker ps

## Verificar la base de datos y la tabla
docker exec -it consultorio-db psql -U postgres -d consultorio -c "SELECT * FROM servicios;"

## Si necesitás detener y eliminar el contenedor
docker stop consultorio-db
docker rm consultorio-db

## Si necesitás reconstruir la imagen desde cero
docker stop consultorio-db
docker rm consultorio-db
docker rmi consultorio-pg
docker build -t consultorio-pg .
docker run --name consultorio-db -e POSTGRES_PASSWORD=1234 -p 5432:5432 -d consultorio-pggit add .
